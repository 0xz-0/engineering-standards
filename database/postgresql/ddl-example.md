# PostgreSQL 完整 DDL 示例

> 符合[通用规范](../naming.md)、[表设计规范](../table-design.md)、[索引规范](../indexing.md)及 PG 专项规范的可执行示例。

```sql
-- ============================================================
-- 1. 订单主表（含冗余字段 + 乐观锁 + 软删除）
-- ============================================================
CREATE TABLE order_info (
    id bigint PRIMARY KEY GENERATED BY DEFAULT AS IDENTITY,
    user_id bigint NOT NULL,
    user_nickname text NOT NULL,
    order_no text NOT NULL,
    amount_cents bigint NOT NULL,
    order_status text NOT NULL DEFAULT 'INIT',
    created_at timestamptz NOT NULL DEFAULT now(),
    updated_at timestamptz NOT NULL DEFAULT now(),
    deleted_at timestamptz DEFAULT NULL,
    version int NOT NULL DEFAULT 0
);

COMMENT ON TABLE order_info IS '订单主表，存储用户下单的核心订单数据。';
COMMENT ON COLUMN order_info.id IS '自增主键（内部使用，禁止对外暴露）。';
COMMENT ON COLUMN order_info.user_id IS '关联 user_info.id；下单用户ID。';
COMMENT ON COLUMN order_info.user_nickname IS '[冗余] 来源：user_info.nickname；下单时用户名快照，避免列表页联表查询。';
COMMENT ON COLUMN order_info.order_no IS '订单编号，用户侧展示的唯一业务编号（TEXT类型）。';
COMMENT ON COLUMN order_info.amount_cents IS '订单总金额（单位：分，bigint）。';
COMMENT ON COLUMN order_info.order_status IS '订单状态。枚举：INIT=待支付, PAID=已支付, SHIPPED=已发货, COMPLETED=已完成, CANCELED=已取消。';
COMMENT ON COLUMN order_info.created_at IS '订单创建时间（UTC）。';
COMMENT ON COLUMN order_info.updated_at IS '订单最后更新时间（UTC，由应用层维护）。';
COMMENT ON COLUMN order_info.deleted_at IS '软删除时间（NULL表示未删除）。';
COMMENT ON COLUMN order_info.version IS '乐观锁版本号（更新时递增）。';

-- 索引
-- user_id 已被下面复合索引的最左列覆盖，不单独建 idx__order_info__user_id
CREATE UNIQUE INDEX uniq__order_info__order_no
    ON order_info(order_no) WHERE deleted_at IS NULL;   -- 软删除场景用部分唯一索引
CREATE INDEX idx__order_info__user_id__created_at
    ON order_info(user_id, created_at);

-- ============================================================
-- 2. 订单商品明细表（带业务属性的关联表，不加 _relation）
-- ============================================================
CREATE TABLE order_product (
    id bigint PRIMARY KEY GENERATED BY DEFAULT AS IDENTITY,
    order_id bigint NOT NULL,
    product_id bigint NOT NULL,
    quantity int NOT NULL,
    unit_price_cents bigint NOT NULL,
    created_at timestamptz NOT NULL DEFAULT now(),
    updated_at timestamptz NOT NULL DEFAULT now(),
    deleted_at timestamptz DEFAULT NULL,
    version int NOT NULL DEFAULT 0
);

COMMENT ON TABLE order_product IS '订单商品明细表，存储订单中包含的商品及购买时快照信息。';
COMMENT ON COLUMN order_product.order_id IS '关联 order_info.id；所属订单ID。';
COMMENT ON COLUMN order_product.product_id IS '关联 product.id；商品ID。';
COMMENT ON COLUMN order_product.quantity IS '购买数量。';
COMMENT ON COLUMN order_product.unit_price_cents IS '下单时商品单价（单位：分，bigint），历史快照。';

CREATE INDEX idx__order_product__order_id ON order_product(order_id);
CREATE INDEX idx__order_product__product_id ON order_product(product_id);

-- ============================================================
-- 3. 纯关联关系表（硬删除 + 仅 created_at + 联合唯一索引防重复）
-- ============================================================
CREATE TABLE user_role_relation (
    id bigint PRIMARY KEY GENERATED BY DEFAULT AS IDENTITY,
    user_id bigint NOT NULL,
    role_id bigint NOT NULL,
    created_at timestamptz NOT NULL DEFAULT now()
);

COMMENT ON TABLE user_role_relation IS '用户与角色的关联关系表（纯关联，硬删除）。';
COMMENT ON COLUMN user_role_relation.user_id IS '关联 user_info.id；用户ID。';
COMMENT ON COLUMN user_role_relation.role_id IS '关联 role.id；角色ID。';
COMMENT ON COLUMN user_role_relation.created_at IS '关联关系建立时间（UTC）。';

-- 联合唯一索引防并发重复关联（禁用外键/CHECK 后的兜底手段），同时覆盖 user_id 正向查询
CREATE UNIQUE INDEX uniq__user_role_relation__user_id__role_id
    ON user_role_relation(user_id, role_id);
-- 反向查询索引：按角色查用户列表（前导列不同，与上方唯一索引不构成冗余）
CREATE INDEX idx__user_role_relation__role_id ON user_role_relation(role_id);
```
